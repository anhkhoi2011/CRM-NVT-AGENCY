'use client';

import React, { useState } from 'react';
import Link from 'next/link';
import { usePathname } from 'next/navigation';
import {
  Wallet,
  User as UserIcon,
  Menu,
  X,
  ShieldCheck,
  LogOut,
  PlusCircle,
  Sparkles,
  ShoppingBag,
  Code2,
  Share2,
  Wrench,
  BookOpen
} from 'lucide-react';
import { useAppStore } from '@/lib/store';
import { formatCurrency } from '@/lib/utils';

interface HeaderProps {
  onOpenAuth: () => void;
  onOpenTopup: () => void;
}

export default function Header({ onOpenAuth, onOpenTopup }: HeaderProps) {
  const pathname = usePathname();
  const { user, logout } = useAppStore();
  const [mobileMenuOpen, setMobileMenuOpen] = useState(false);
  const [userDropdownOpen, setUserDropdownOpen] = useState(false);

  const navLinks = [
    { name: 'Dịch vụ mạng xã hội', href: '/dich-vu-mang-xa-hoi', icon: Share2 },
    { name: 'Code & Tool', href: '/code-tool', icon: Code2 },
    { name: 'Công cụ AI', href: '/cong-cu-ai', icon: Sparkles },
    { name: 'Tài Khoản', href: '/tai-khoan', icon: ShoppingBag },
    { name: 'Tiện ích', href: '/tien-ich', icon: Wrench },
    { name: 'Blog', href: '/blog', icon: BookOpen },
  ];

  return (
    <header className="sticky top-0 z-50 bg-white/95 backdrop-blur-md border-b border-borderSoft shadow-sm">
      <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8">
        <div className="flex items-center justify-between h-20 gap-4">
          
          {/* LOGO */}
          <Link href="/" className="flex items-center gap-3 flex-shrink-0 group">
            <div className="w-11 h-11 rounded-2xl bg-gradient-to-tr from-brand-700 via-brand-500 to-cyan-400 flex items-center justify-center text-white shadow-md shadow-brand-500/20 group-hover:scale-105 transition-transform duration-300">
              <span className="font-extrabold text-xl tracking-tighter">TC</span>
            </div>
            <div>
              <div className="font-black text-xl tracking-tight text-navy-900 group-hover:text-brand-600 transition-colors">
                TRẠM CÔNG NGHỆ
              </div>
              <div className="text-[10px] uppercase font-bold tracking-widest text-brand-600">
                Ecosystem Platform
              </div>
            </div>
          </Link>

          {/* DESKTOP NAV */}
          <nav className="hidden xl:flex items-center gap-1">
            {navLinks.map((link) => {
              const isActive = pathname === link.href || pathname.startsWith(`${link.href}/`);
              return (
                <Link
                  key={link.href}
                  href={link.href}
                  className={`px-3.5 py-2 rounded-xl text-sm font-medium transition-all duration-200 ${
                    isActive
                      ? 'text-brand-600 bg-brand-50 font-semibold'
                      : 'text-slate-700 hover:text-brand-600 hover:bg-slate-50'
                  }`}
                >
                  {link.name}
                </Link>
              );
            })}
          </nav>

          {/* USER ACTIONS & AUTH */}
          <div className="flex items-center gap-3">
            {user ? (
              <div className="flex items-center gap-3">
                {/* WALLET BUTTON */}
                <button
                  onClick={onOpenTopup}
                  className="flex items-center gap-2 bg-gradient-to-r from-brand-50 to-cyan-50 border border-brand-200 px-3.5 py-1.5 rounded-2xl hover:border-brand-400 transition-all group"
                  title="Nạp thêm số dư"
                >
                  <div className="w-7 h-7 rounded-xl bg-brand-500 text-white flex items-center justify-center">
                    <Wallet className="w-4 h-4" />
                  </div>
                  <div className="text-left hidden sm:block">
                    <div className="text-[10px] text-slate-500 font-medium leading-none">Số dư tài khoản</div>
                    <div className="text-xs font-bold text-brand-700">{formatCurrency(user.balance)}</div>
                  </div>
                  <PlusCircle className="w-4 h-4 text-brand-500 group-hover:scale-110 transition-transform ml-1" />
                </button>

                {/* USER DROPDOWN */}
                <div className="relative">
                  <button
                    onClick={() => setUserDropdownOpen(!userDropdownOpen)}
                    className="flex items-center gap-2 p-1.5 rounded-2xl hover:bg-slate-100 transition-colors"
                  >
                    <img
                      src={user.avatarUrl || 'https://avatar.vercel.sh/user'}
                      alt={user.name}
                      className="w-9 h-9 rounded-xl object-cover border border-slate-200"
                    />
                  </button>

                  {userDropdownOpen && (
                    <div className="absolute right-0 mt-2 w-64 bg-white rounded-2xl shadow-xl border border-slate-100 p-2 z-50">
                      <div className="p-3 border-b border-slate-100">
                        <div className="font-bold text-slate-900 truncate">{user.name}</div>
                        <div className="text-xs text-slate-500">{user.phone}</div>
                        <div className="mt-1.5 inline-block text-[10px] font-bold px-2 py-0.5 bg-brand-100 text-brand-700 rounded-full">
                          {user.role}
                        </div>
                      </div>

                      <div className="py-1 space-y-0.5">
                        <Link
                          href="/account"
                          onClick={() => setUserDropdownOpen(false)}
                          className="flex items-center gap-2.5 px-3 py-2 text-sm text-slate-700 hover:bg-slate-50 rounded-xl"
                        >
                          <UserIcon className="w-4 h-4 text-slate-400" />
                          Dashboard Cá Nhân
                        </Link>
                        <Link
                          href="/account/orders"
                          onClick={() => setUserDropdownOpen(false)}
                          className="flex items-center gap-2.5 px-3 py-2 text-sm text-slate-700 hover:bg-slate-50 rounded-xl"
                        >
                          <ShoppingBag className="w-4 h-4 text-slate-400" />
                          Lịch Sử Đơn Hàng
                        </Link>
                      {(user.role === 'SUPER_ADMIN' || user.role === 'ADMIN') && (
                        <Link
                          href="/admin"
                          onClick={() => setUserDropdownOpen(false)}
                          className="flex items-center gap-2.5 px-3 py-2 text-sm font-semibold text-purple-700 hover:bg-purple-50 rounded-xl"
                        >
                          <ShieldCheck className="w-4 h-4 text-purple-600" />
                          Admin Dashboard
                        </Link>
                      )}
                      </div>

                      <div className="pt-1 border-t border-slate-100">
                        <button
                          onClick={() => {
                            logout();
                            setUserDropdownOpen(false);
                          }}
                          className="w-full flex items-center gap-2.5 px-3 py-2 text-sm text-rose-600 hover:bg-rose-50 rounded-xl font-medium"
                        >
                          <LogOut className="w-4 h-4 text-rose-500" />
                          Đăng Xuất
                        </button>
                      </div>
                    </div>
                  )}
                </div>
              </div>
            ) : (
              <button
                onClick={onOpenAuth}
                className="bg-gradient-to-r from-brand-600 to-cyan-600 hover:from-brand-700 hover:to-cyan-700 text-white font-semibold text-sm px-5 py-2.5 rounded-2xl shadow-md shadow-brand-500/20 hover:shadow-lg transition-all"
              >
                Đăng Nhập / Đăng Ký
              </button>
            )}

            {/* MOBILE HAMBURGER */}
            <button
              onClick={() => setMobileMenuOpen(!mobileMenuOpen)}
              className="xl:hidden p-2 rounded-xl text-slate-700 hover:bg-slate-100"
            >
              {mobileMenuOpen ? <X className="w-6 h-6" /> : <Menu className="w-6 h-6" />}
            </button>
          </div>
        </div>

        {/* MOBILE MENU ACCORDION */}
        {mobileMenuOpen && (
          <div className="xl:hidden py-4 border-t border-slate-100 space-y-2">
            {navLinks.map((link) => (
              <Link
                key={link.href}
                href={link.href}
                onClick={() => setMobileMenuOpen(false)}
                className="flex items-center gap-3 px-4 py-2.5 text-slate-700 hover:bg-brand-50 hover:text-brand-600 rounded-xl font-medium"
              >
                <link.icon className="w-5 h-5 text-slate-400" />
                {link.name}
              </Link>
            ))}
          </div>
        )}
      </div>
    </header>
  );
}
