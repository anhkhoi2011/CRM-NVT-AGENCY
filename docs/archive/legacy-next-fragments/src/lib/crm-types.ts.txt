/**
 * CRM — Kiểu dữ liệu trung tâm cho Customer Status (spec #54).
 * Vị trí thật trong project: src/lib/crm-types.ts
 *
 * Nguyên tắc: MỘT nguồn dữ liệu duy nhất cho mọi Role (54.15).
 * Không tạo bảng/dữ liệu riêng cho Sale / Leader / Admin.
 */

/** Role dùng trong CRM. types.ts gốc chỉ có SUPER_ADMIN | ADMIN | USER — xem CRM-README.md để thêm LEADER | SALE. */
export type CrmRole = 'SUPER_ADMIN' | 'ADMIN' | 'LEADER' | 'SALE' | 'USER';

/** Bản chất nghiệp vụ của trạng thái — dùng cho thống kê, KHÔNG dùng để hard-code logic UI. */
export type StatusKind = 'PIPELINE' | 'WON' | 'LOST' | 'NEUTRAL';

/** 54.1 + 54.14 — bảng customer_statuses. Admin cấu hình hoàn toàn, không hard-code. */
export interface CustomerStatus {
  id: string;
  code: string;
  name: string;
  description?: string;
  /** Màu badge (hex) — thuộc cấu hình của Admin, không phải màu biểu đồ. */
  color: string;
  kind: StatusKind;
  /** Có xuất hiện trong biểu đồ Funnel hay không (54.9). */
  isFunnelStage: boolean;
  sortOrder: number;
  isActive: boolean;
  /** 54.13 — bật notification cho Leader/Admin khi khách chuyển sang trạng thái này. */
  notifyOnChange: boolean;
  /** Trạng thái hệ thống (VD: "Khác") — không cho xóa. */
  isSystem?: boolean;
  createdAt: string;
  updatedAt: string;
}

/** 54.14 — Customer chỉ lưu currentStatusId, toàn bộ lịch sử nằm ở customer_status_history. */
export interface Customer {
  id: string;
  code: string;
  name: string;
  phone: string;
  email?: string;
  /** Nguồn lead: Landing Page, TikTok Ads, Zalo... */
  source: string;
  /** SĐT của Sale phụ trách (null = chưa phân bổ). */
  assignedSalePhone: string | null;
  currentStatusId: string;
  statusUpdatedAt: string;
  /** 54.11 — liên kết sang tài khoản nền tảng để đối chiếu Order/Payment, nhưng trạng thái độc lập. */
  linkedUserPhone?: string;
  tags?: string[];
  /** Ghi chú mới nhất (snapshot) — bản đầy đủ nằm trong timeline. */
  lastNote?: string;
  createdAt: string;
  createdBy: string;
}

/** 54.5 + 54.14 — bảng customer_status_history. Append-only: không có hàm sửa/xóa. */
export interface CustomerStatusHistory {
  id: string;
  customerId: string;
  oldStatusId: string | null;
  newStatusId: string;
  changedBy: string;
  changedByName: string;
  changedByRole: CrmRole;
  note?: string;
  ip?: string;
  channel: 'WEB' | 'API' | 'SYSTEM';
  createdAt: string;
}

/** 54.6 — Customer Timeline gộp nhiều loại sự kiện. */
export type TimelineKind =
  | 'STATUS'
  | 'ASSIGNMENT'
  | 'NOTE'
  | 'ORDER'
  | 'PAYMENT'
  | 'API'
  | 'ACTIVITY'
  | 'SYSTEM';

export interface TimelineEvent {
  id: string;
  customerId: string;
  kind: TimelineKind;
  title: string;
  detail?: string;
  actorName: string;
  actorRole: CrmRole | 'SYSTEM';
  createdAt: string;
  /** Màu badge lấy từ CustomerStatus khi kind === 'STATUS'. */
  statusColor?: string;
}

/** Team bán hàng: Leader ↔ danh sách Sale (dùng cho phân quyền 54.8). */
export interface SaleTeam {
  id: string;
  name: string;
  leaderPhone: string;
  memberPhones: string[];
}

/** 54.13 — Notification cho Leader/Admin. */
export interface CrmNotification {
  id: string;
  recipientPhone: string;
  title: string;
  body: string;
  customerId?: string;
  statusId?: string;
  read: boolean;
  createdAt: string;
}

/** 54.12 bước 7 — Audit log cho mọi thay đổi. */
export interface CrmAuditLog {
  id: string;
  actorPhone: string;
  actorRole: CrmRole;
  action: string;
  entity: 'customer' | 'customer_status' | 'customer_status_history';
  entityId: string;
  before?: string;
  after?: string;
  ip?: string;
  userAgent?: string;
  channel: 'WEB' | 'API' | 'SYSTEM';
  createdAt: string;
}

/** Cấu hình CRM do Admin bật/tắt. */
export interface CrmSettings {
  /** 54.8 — Leader chỉ được cập nhật trạng thái nếu Admin cấp quyền. */
  leaderCanUpdateStatus: boolean;
  /** 54.13 — gửi notification cho toàn bộ Admin khi đạt trạng thái quan trọng. */
  notifyAdminsOnMilestone: boolean;
  /** 54.4 — cơ chế real-time đang dùng. */
  realtimeMode: 'BROADCAST' | 'POLL' | 'OFF';
}

export interface CrmActionResult {
  success: boolean;
  message: string;
  /** Mã lỗi để API trả HTTP status tương ứng. */
  code?: 'UNAUTHENTICATED' | 'FORBIDDEN' | 'NOT_FOUND' | 'INVALID_STATUS' | 'IN_USE' | 'NO_CHANGE' | 'OK';
}

/** Sự kiện real-time phát qua BroadcastChannel / SSE (54.4). */
export interface CrmRealtimeEvent {
  id: string;
  type: 'STATUS_CHANGED' | 'NOTE_ADDED' | 'CUSTOMER_ASSIGNED' | 'STATUS_CONFIG_CHANGED';
  customerId?: string;
  customerName?: string;
  statusName?: string;
  statusColor?: string;
  actorName?: string;
  actorRole?: CrmRole;
  at: string;
}
