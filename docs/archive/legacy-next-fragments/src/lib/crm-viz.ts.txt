/** Series đơn — mọi thanh cùng một màu (identity do tiêu đề chart mang). */
export const VIZ_SERIES_1 = { light: '#2a78d6', dark: '#3987e5' };

/** Series thứ 2 khi cần so sánh 2 đường (VD: cập nhật hôm nay vs trung bình). */
export const VIZ_SERIES_2 = { light: '#eb6834', dark: '#d95926' };

/** Sequential blue 100→700 — dùng cho heatmap (magnitude liên tục). */
export const VIZ_SEQUENTIAL = [
  '#cde2fb', '#9ec5f4', '#6da7ec', '#3987e5', '#256abf', '#184f95', '#0d366b',
];

/** Status token (cố định, luôn kèm icon + nhãn chữ — không bao giờ chỉ dựa vào màu). */
export const VIZ_STATUS = {
  good: '#0ca30c',
  warning: '#fab219',
  serious: '#ec835a',
  critical: '#d03b3b',
};

/** Ink & chrome — text KHÔNG BAO GIỜ mang màu series. */
export const VIZ_INK = {
  grid: { light: '#e1e0d9', dark: '#2c2c2a' },
  axis: { light: '#c3c2b7', dark: '#383835' },
  muted: '#898781',

/**
 * Chọn bước sequential theo tỉ lệ 0..1 cho ô heatmap.
 * Trả về { bg, fg } — chữ trong ô chọn trắng/mực theo độ sáng nền (ngoại lệ hợp lệ
 * duy nhất cho việc đặt text trên vùng màu).
 */
export function seqStep(ratio: number): { bg: string; fg: string; index: number } {
  if (ratio <= 0) return { bg: 'transparent', fg: 'currentColor', index: -1 };
  const idx = Math.min(VIZ_SEQUENTIAL.length - 1, Math.max(0, Math.round(ratio * (VIZ_SEQUENTIAL.length - 1))));
  return { bg: VIZ_SEQUENTIAL[idx], fg: idx >= 3 ? '#ffffff' : '#0b0b0b', index: idx };
}
