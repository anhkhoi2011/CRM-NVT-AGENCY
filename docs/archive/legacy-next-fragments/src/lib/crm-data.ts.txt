/* ---------- Sinh 320 khách hàng mô phỏng (deterministic — không dùng Math.random để tránh hydration mismatch) ---------- */

const FIRST = ['Nguyễn', 'Trần', 'Lê', 'Phạm', 'Hoàng', 'Vũ', 'Đặng', 'Bùi', 'Đỗ', 'Ngô', 'Dương', 'Lý'];

const SOURCES = ['Landing Page A', 'Landing Page B', 'TikTok Ads', 'Facebook Ads', 'Google Search', 'Zalo OA', 'Giới thiệu', 'Hotline'];
const SALE_PHONES = ['0966000001', '0966000002', '0966000003', '0966000004', '0966000005'];

/** Phân bổ trạng thái theo hình phễu: nhiều lead đầu phễu, ít khách chốt. */
const STATUS_WEIGHTS: { id: string; w: number }[] = [
  { id: 'st-01', w: 46 }, { id: 'st-02', w: 34 }, { id: 'st-03', w: 28 }, { id: 'st-04', w: 20 },
  { id: 'st-05', w: 16 }, { id: 'st-06', w: 12 }, { id: 'st-07', w: 9 }, { id: 'st-08', w: 7 },
  { id: 'st-09', w: 11 }, { id: 'st-10', w: 24 }, { id: 'st-11', w: 10 }, { id: 'st-12', w: 8 },
  { id: 'st-13', w: 13 }, { id: 'st-14', w: 5 }, { id: 'st-15', w: 3 },
];

const WEIGHT_POOL: string[] = STATUS_WEIGHTS.flatMap(s => Array.from({ length: s.w }, () => s.id));

/** Chuỗi trạng thái đi qua trước khi tới trạng thái hiện tại (54.5 — lịch sử phải hợp lý). */
const PIPELINE_ORDER = ['st-01', 'st-02', 'st-03', 'st-04', 'st-05', 'st-06', 'st-07', 'st-08'];

function pad(n: number, len = 2) {
  return String(n).padStart(len, '0');
}

/** Ngày giờ dạng dd/MM/yyyy HH:mm — lùi `daysAgo` ngày kể từ 04/09/2026. */
function stamp(daysAgo: number, hour: number, minute: number): string {
  const base = Date.UTC(2026, 8, 4, 12, 0, 0);
  const d = new Date(base - daysAgo * 86400000);
  return `${pad(d.getUTCDate())}/${pad(d.getUTCMonth() + 1)}/${d.getUTCFullYear()} ${pad(hour)}:${pad(minute)}`;
}

const CUSTOMERS: Customer[] = [];
const HISTORY: CustomerStatusHistory[] = [];

for (let i = 0; i < 320; i++) {
  const name = `${FIRST[i % FIRST.length]} ${MID[(i * 3) % MID.length]} ${LAST[(i * 7) % LAST.length]}`;
  const statusId = WEIGHT_POOL[(i * 13) % WEIGHT_POOL.length];
  const salePhone = i % 17 === 0 ? null : SALE_PHONES[(i * 3) % SALE_PHONES.length];
  const daysAgo = 1 + ((i * 11) % 60);
  const createdAt = stamp(daysAgo, 8 + (i % 9), (i * 7) % 60);
  const id = `cus-${pad(i + 1, 4)}`;

  // Chuỗi trạng thái đã đi qua: từ Lead mới → ... → trạng thái hiện tại
  const idxInPipeline = PIPELINE_ORDER.indexOf(statusId);
  const path = idxInPipeline >= 0
    ? PIPELINE_ORDER.slice(0, idxInPipeline + 1)
    : ['st-01', 'st-02', statusId];

  let statusUpdatedAt = createdAt;
  let prev: string | null = null;

  path.forEach((sid, step) => {
    const at = stamp(Math.max(daysAgo - step, 0), 9 + ((i + step) % 8), ((i * 5) + step * 17) % 60);
    statusUpdatedAt = at;
    HISTORY.push({
      id: `hist-${id}-${step}`,
      customerId: id,
      oldStatusId: prev,
      newStatusId: sid,
      changedBy: step === 0 ? 'system' : (salePhone || 'system'),
      changedByName: step === 0 ? 'Hệ thống' : (CRM_STAFF.find(s => s.phone === salePhone)?.name || 'Hệ thống'),
      changedByRole: step === 0 ? 'SUPER_ADMIN' : (salePhone ? 'SALE' : 'SUPER_ADMIN'),
      note: step === 0 ? `Lead được tạo từ ${SOURCES[i % SOURCES.length]}` : undefined,
      channel: step === 0 ? 'SYSTEM' : 'WEB',
      createdAt: at,
    });
    prev = sid;
  });

  CUSTOMERS.push({
    id,
    code: `KH${pad(i + 1, 5)}`,
    name,
    phone: `09${pad(10 + (i % 80), 2)}${pad(100000 + i * 7, 6)}`.slice(0, 10),
    email: i % 3 === 0 ? `khach${i + 1}@gmail.com` : undefined,
    source: SOURCES[i % SOURCES.length],
    assignedSalePhone: salePhone,
    currentStatusId: statusId,
    statusUpdatedAt,
    tags: i % 5 === 0 ? ['VIP'] : undefined,
    createdAt,
    createdBy: 'system',
  });
}

export const CRM_CUSTOMERS = CUSTOMERS;
export const CRM_STATUS_HISTORY = HISTORY;
