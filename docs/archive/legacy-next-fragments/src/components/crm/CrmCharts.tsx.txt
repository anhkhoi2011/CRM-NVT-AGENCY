'use client';

/**
 * CRM — Biểu đồ cho 54.9 (Status Distribution / Funnel / Conversion) và 54.10 (Báo cáo Sale).
 * Vị trí thật trong project: src/components/crm/CrmCharts.tsx
 *
 * Quy tắc dataviz đã áp dụng (đã chạy validator, xem crm-viz.ts):
 *  - Một series → MỘT màu (slot-1 blue) cho mọi thanh. Không tô đậm-nhạt theo giá trị
 *    (độ dài thanh đã mang thông tin đó rồi).
 *  - Thanh ≤ 24px, bo 4px ở đầu dữ liệu, vuông ở chân trục, cách nhau ≥ 2px.
 *  - Nhãn giá trị đặt ở đầu thanh (direct label) → không phụ thuộc tooltip.
 *  - Mọi chart đều có "Xem dạng bảng" (table view twin).
 *  - Heatmap dùng ramp 1 hue + SỐ trong ô → không bao giờ chỉ dựa vào màu.
 *  - Grid/axis: hairline 1px liền, màu lùi về sau.
 */

import React, { useMemo, useState } from 'react';
import { Table2, BarChart3 } from 'lucide-react';
import { CustomerStatus } from '@/lib/crm-types';
import { VIZ_SERIES_1, VIZ_SEQUENTIAL, seqStep } from '@/lib/crm-viz';
import { crmCard } from './CrmPrimitives';

const fmt = (n: number) => n.toLocaleString('vi-VN');
const pct1 = (n: number) => `${n.toFixed(1).replace('.', ',')}%`;

/** Nút chuyển chart ↔ bảng. Bảng là bản đọc được không cần màu. */
function ViewToggle({ view, onChange }: { view: 'chart' | 'table'; onChange: (v: 'chart' | 'table') => void }) {
  return (
    <div className="flex bg-slate-100 dark:bg-slate-800 rounded-xl p-1">
      {(['chart', 'table'] as const).map(v => (
        <button
          key={v}
          onClick={() => onChange(v)}
          aria-pressed={view === v}
          className={`flex items-center gap-1.5 px-3 py-1.5 rounded-lg text-[11px] font-extrabold transition-all ${
            view === v
              ? 'bg-white dark:bg-slate-700 text-brand-600 dark:text-cyan-400 shadow-sm'
              : 'text-slate-500 dark:text-slate-400 hover:text-slate-700 dark:hover:text-slate-200'}`}
        >
          {v === 'chart' ? <BarChart3 className="w-3.5 h-3.5" /> : <Table2 className="w-3.5 h-3.5" />}
          {v === 'chart' ? 'Biểu đồ' : 'Bảng'}
        </button>
      ))}
    </div>
  );
}

/* ================= 54.9a — STATUS DISTRIBUTION (thanh ngang, 1 series) ================= */

export function StatusDistributionChart({ data, total, title, subtitle }: {
  data: { status: CustomerStatus; count: number }[];
  total: number;
  title: string;
  subtitle?: string;
}) {
  const [view, setView] = useState<'chart' | 'table'>('chart');
  const [hover, setHover] = useState<string | null>(null);

  const rows = useMemo(() => data.filter(d => d.count > 0), [data]);
  const max = Math.max(1, ...rows.map(r => r.count));

  return (
    <section className={`${crmCard} p-5 sm:p-6 space-y-5`}>
      <header className="flex items-start justify-between gap-3 flex-wrap">
        <div>
          <h3 className="font-extrabold text-navy-900 dark:text-white">{title}</h3>
          <p className="text-xs text-slate-400 dark:text-slate-500 mt-0.5">
            {subtitle || `${fmt(total)} khách hàng · ${rows.length} trạng thái đang có khách`}
          </p>
        </div>
        <ViewToggle view={view} onChange={setView} />
      </header>

      {rows.length === 0 ? (
        <p className="text-sm text-slate-400 py-8 text-center">Chưa có khách hàng nào trong phạm vi của bạn.</p>
      ) : view === 'chart' ? (
        <div className="space-y-2.5">
          {rows.map(({ status, count }) => {
            const w = (count / max) * 100;
            const share = total ? (count / total) * 100 : 0;
            const isHover = hover === status.id;
            return (
              <div
                key={status.id}
                className="grid grid-cols-[minmax(96px,180px)_1fr] items-center gap-3 py-1 cursor-default"
                onMouseEnter={() => setHover(status.id)}
                onMouseLeave={() => setHover(null)}
              >
                {/* Nhãn: chấm màu + chữ. Chữ dùng token text, không mang màu series. */}
                <div className="flex items-center gap-2 min-w-0">
                  <span className="w-2 h-2 rounded-full flex-shrink-0" style={{ background: status.color }} aria-hidden />
                  <span className="text-xs font-semibold text-slate-600 dark:text-slate-300 truncate" title={status.name}>
                    {status.name}
                  </span>
                </div>

                <div className="flex items-center gap-2.5 min-w-0">
                  <div className="flex-1 h-5 relative">
                    <div
                      className="absolute inset-y-0 left-0 transition-all duration-300"
                      style={{
                        width: `${Math.max(w, 1.2)}%`,
                        background: VIZ_SERIES_1.light,
                        borderRadius: '0 4px 4px 0',
                        opacity: isHover ? 1 : 0.92,
                      }}
                    />
                  </div>
                  {/* Direct label ở đầu thanh — giá trị không bị khoá sau tooltip. */}
                  <div className="flex items-baseline gap-1.5 w-[112px] flex-shrink-0 justify-end">
                    <span className="text-xs font-extrabold text-navy-900 dark:text-white tabular-nums">{fmt(count)}</span>
                    <span className="text-[10px] font-semibold text-slate-400 dark:text-slate-500 tabular-nums">{pct1(share)}</span>
                  </div>
                </div>
              </div>
            );
          })}
        </div>
      ) : (
        <div className="overflow-x-auto">
          <table className="w-full text-sm">
            <caption className="sr-only">{title} — dạng bảng</caption>
            <thead className="bg-slate-50 dark:bg-slate-800/60">
              <tr>
                <th scope="col" className="px-3 py-2 text-left text-[10px] font-extrabold uppercase tracking-wider text-slate-400">Trạng thái</th>
                <th scope="col" className="px-3 py-2 text-right text-[10px] font-extrabold uppercase tracking-wider text-slate-400">Số khách</th>
                <th scope="col" className="px-3 py-2 text-right text-[10px] font-extrabold uppercase tracking-wider text-slate-400">Tỷ trọng</th>
              </tr>
            </thead>
            <tbody>
              {data.map(({ status, count }) => (
                <tr key={status.id} className="border-t border-slate-100 dark:border-slate-800">
                  <td className="px-3 py-2 text-slate-700 dark:text-slate-200 font-semibold">{status.name}</td>
                  <td className="px-3 py-2 text-right font-extrabold text-navy-900 dark:text-white tabular-nums">{fmt(count)}</td>
                  <td className="px-3 py-2 text-right text-slate-500 dark:text-slate-400 tabular-nums">
                    {total ? pct1((count / total) * 100) : '—'}
                  </td>
                </tr>
              ))}
              <tr className="border-t-2 border-slate-200 dark:border-slate-700 bg-slate-50 dark:bg-slate-800/60">
                <td className="px-3 py-2 font-extrabold text-navy-900 dark:text-white">Tổng</td>
                <td className="px-3 py-2 text-right font-extrabold text-navy-900 dark:text-white tabular-nums">{fmt(total)}</td>
                <td className="px-3 py-2 text-right text-slate-500 tabular-nums">100%</td>
              </tr>
            </tbody>
          </table>
        </div>
      )}
    </section>
  );
}
