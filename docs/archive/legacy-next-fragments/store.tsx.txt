const MOCK_DAILY_STATS: DailyStat[] = Array.from({ length: 28 }, (_, i) => {
  const day = i + 1;
  const weekendDip = day % 7 === 0 || day % 7 === 6 ? 90 : 0;
  const visits = 300 + day * 11 + ((day * 7) % 5) * 22 - weekendDip;
  const revenue = visits * (5100 + ((day * 13) % 7) * 420);
  return {
    date: `${String(day).padStart(2, '0')}/08`,
    visits,
    revenue,
  };
});

  };

  const deleteSocialService = (platformId: string, serviceId: string) => {
    setSocialPlatforms(prev => prev.map(p => p.id === platformId ? {
      ...p,
      services: p.services.filter(s => s.id !== serviceId)
    } : p));
  };

  const addCodeProduct = (product: any) => {
    setCodeProducts(prev => [product, ...prev]);
  };

  const updateCodeProduct = (id: string, data: any) => {
    setCodeProducts(prev => prev.map(p => p.id === id ? { ...p, ...data } : p));
  };

  const deleteCodeProduct = (id: string) => {
    setCodeProducts(prev => prev.filter(p => p.id !== id));
  };

  const addAiTemplate = (template: any) => {
    setAiTemplates(prev => [template, ...prev]);
  };

  const updateAiTemplate = (id: string, data: any) => {
    setAiTemplates(prev => prev.map(t => t.id === id ? { ...t, ...data } : t));
  };

  const deleteAiTemplate = (id: string) => {
    setAiTemplates(prev => prev.filter(t => t.id !== id));
  };

  const addAccountProduct = (product: any) => {
    setAccountProducts(prev => [product, ...prev]);
  };

  const updateAccountProduct = (id: string, data: any) => {
    setAccountProducts(prev => prev.map(p => p.id === id ? { ...p, ...data } : p));
  };

  const deleteAccountProduct = (id: string) => {
    setAccountProducts(prev => prev.filter(p => p.id !== id));
  };

  const addBlogPost = (post: any) => {
    setBlogPosts(prev => [post, ...prev]);
  };

  const updateBlogPost = (id: string, data: any) => {
    setBlogPosts(prev => prev.map(p => p.id === id ? { ...p, ...data } : p));
  };

  const deleteBlogPost = (id: string) => {
    setBlogPosts(prev => prev.filter(p => p.id !== id));
  };

  return (
    <AppContext.Provider value={{
      user,
      orders,
      transactions,
      aiJobs,
      tickets,
      emergencyMode,
      banners,
      allUsers,
      dailyStats,
      monthlyStats: MONTHLY_STATS,
      trafficSources: TRAFFIC_SOURCES,
      pageVisits,
      socialPlatforms,
      codeProducts,
      aiTemplates,
      accountProducts,
      blogPosts,
      login,
      logout,
      topup,
      createOrder,
      createAiJob,
      approveCodeOrder,
      updateOrderStatus,
      toggleEmergencyMode,
      updateBanner,
      createTicket,
      promoteToAdmin,
      demoteToUser,
      adjustUserBalance,
      deleteUser,
      incrementPageVisits,
      addSocialService,
      updateSocialService,
      deleteSocialService,
      addCodeProduct,
      updateCodeProduct,
      deleteCodeProduct,
      addAiTemplate,
      updateAiTemplate,
      deleteAiTemplate,
      addAccountProduct,
      updateAccountProduct,
      deleteAccountProduct,
      addBlogPost,
      updateBlogPost,
      deleteBlogPost,
    }}>
      {children}
    </AppContext.Provider>
  );
}

export function useAppStore() {
  const context = useContext(AppContext);
  if (!context) {
    throw new Error('useAppStore must be used within an AppProvider');
  }
  return context;
}
